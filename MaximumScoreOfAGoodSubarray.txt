//approach - 1
//time complexity - O(n)
//space complexity - O(1)
class Solution {
public:
    int maximumScore(vector<int>& nums, int k) {
        int n = nums.size();

        // Initialize two pointers at the required index k
        int i = k;
        int j = k;
        int currMin = nums[k];
        int result = nums[k];

        // Expand the window [i, j] until it covers the entire array
        while(i > 0 || j < n - 1) {
            // Check values of immediate neighbors to the left and right
            int leftValue = (i > 0) ? nums[i - 1] : 0;
            int rightValue = (j < n - 1) ? nums[j + 1] : 0;

            // GREEDY CHOICE: Always expand toward the larger neighbor
            // to keep the minimum value of the window as high as possible
            if(leftValue < rightValue) {
                j++; // Move right
                currMin = min(currMin, nums[j]);
            }
            else {
                i--; // Move left
                currMin = min(currMin, nums[i]);
            }

            // Update result: score = (minimum value in window) * (window length)
            result = max(result, currMin * (j - i + 1));
        }

        return result;
    }
};
