//approach - 2
//time complexityt - O(n)
//space complexity - O(n)
class Solution {
public:
    string predictPartyVictory(string senate) {
        int n = senate.length();
        queue<int> qr, qd;

        // Step 1: Store indices of Radiant and Dire senators in separate queues
        for (int i = 0; i < n; i++) {
            if (senate[i] == 'R') qr.push(i);
            else qd.push(i);
        }

        // Step 2: Compare front-most senators from both parties
        while (!qr.empty() && !qd.empty()) {
            int r_idx = qr.front(); qr.pop();
            int d_idx = qd.front(); qd.pop();

            // The senator with the smaller index acts first and bans the other
            if (r_idx < d_idx) {
                // Radiant wins this round: re-queue with updated index for next round
                qr.push(r_idx + n);
            } else {
                // Dire wins this round: re-queue with updated index for next round
                qd.push(d_idx + n);
            }
        }

        // Step 3: The party that still has senators remaining wins
        return qr.empty() ? "Dire" : "Radiant";
    }
};
