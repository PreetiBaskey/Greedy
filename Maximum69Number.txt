//approach - 2
//time complexity - O(d) - while loop iterates once for each digit in num
//space complexity - O(1)
class Solution {
public:
    int maximum69Number (int num) { 
        int placeValue = 0;
        int placeValueSix = -1;


        int tempNum = num;
        while(tempNum != 0) {
            int remainder = tempNum % 10;
            
            if(remainder == 6) {
                placeValueSix = placeValue;
            }
            
            tempNum = tempNum / 10;
            placeValue++;
        }

        if(placeValueSix == -1) {
            return num;
        }

        return num + 3 * pow(10, placeValueSix);
    }
};
