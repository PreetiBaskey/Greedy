//approach - 1
//time complexity - O(n)
//space complexity - O(1)
class Solution {
public:
    int minCost(string colors, vector<int>& neededTime) {
        int n = colors.size();
        int time = 0;
        int prevMax = 0; // Tracks the maximum removal time in the current group of same colors

        for(int i = 0; i < n; i++) {
            // If the color changes, reset the max tracker for the new group
            if(i > 0 && colors[i] != colors[i - 1]) {
                prevMax = 0;
            }

            int curr = neededTime[i];

            // Greedy Choice: Out of two adjacent same-colored balloons, 
            // always 'remove' the one that takes less time.
            time += min(prevMax, curr);

            // Keep track of the most 'expensive' balloon in the group to avoid removing it
            prevMax = max(prevMax, curr);
        }

        return time;
    }
};
