//approach - 1
//time complexity - O(n)
//space complexity - O(1)
class Solution {
public:
    long long minSum(vector<int>& nums1, vector<int>& nums2) {
        // We use long long to prevent overflow since array sums can be very large
        int countZero1 = 0;
        int countZero2 = 0;
        long long sum1 = 0;
        long long sum2 = 0;

        // Calculate minimum possible sum for nums1 (treating each 0 as 1)
        for(int num : nums1) {
            if(num == 0) {
                countZero1++;
                sum1 += 1; // Minimum value a 0 can take is 1
            } else {
                sum1 += num;
            }
        }

        // Calculate minimum possible sum for nums2 (treating each 0 as 1)
        for(int num : nums2) {
            if(num == 0) {
                countZero2++;
                sum2 += 1; // Minimum value a 0 can take is 1
            } else {
                sum2 += num;
            }
        }

        /* 
         * Logic:
         * 1. If sum1 < sum2 but nums1 has no zeros, we can't increase sum1 to reach sum2.
         * 2. If sum2 < sum1 but nums2 has no zeros, we can't increase sum2 to reach sum1.
         */
        if (sum1 < sum2 && countZero1 == 0) return -1;
        if (sum2 < sum1 && countZero2 == 0) return -1;

        /* 
         * If we can increase the smaller sum (or if they are equal), 
         * the answer is the larger of the two minimum sums.
         */
        return max(sum1, sum2);
    }
};
