//approach - 2 - without using set, with sorting
//time complexity - O(N)
//space complexity - O(1) - O(26)
class Solution {
public:
    int minDeletions(string s) {
        int result = 0;
        int n = s.length();

        vector<int> freq(26, 0);
        for(char &ch: s) {
            freq[ch - 'a']++;
        }

        sort(freq.begin(), freq.end());
        for(int i = 24; i >= 0; i--) {
            if(freq[i] >= freq[i + 1]) {
                int prev = freq[i];
                
                freq[i] = max(0, freq[i + 1] - 1);
                result += prev - freq[i];
            }
        }

        return result;      
    }
};
